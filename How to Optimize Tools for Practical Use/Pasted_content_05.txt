def _find_project_root(extracted: Path) -> Path:
entries = [p for p in extracted.iterdir() if not p.name.startswith(”__MACOSX”)]
dirs = [p for p in entries if p.is_dir()]
files = [p for p in entries if p.is_file()]
if len(dirs) == 1 and not files:
return dirs[0]
return extracted

def _imports_from_file(py: Path) -> List[str]:
try:
tree = ast.parse(py.read_text(encoding=“utf-8”, errors=“ignore”))
except Exception:
return []
imps = []
for node in ast.walk(tree):
if isinstance(node, ast.Import):
for n in node.names:
imps.append(n.name.split(”.”)[0])
elif isinstance(node, ast.ImportFrom):
if node.module:
imps.append(node.module.split(”.”)[0])
return imps

def _infer_requirements(root: Path) -> List[str]:
imports = []
for p in _py_files(root)[:80]:
imports.extend(_imports_from_file(p))
uniq = sorted(set(imports))

```
reqs = []
for name in uniq:
    if name in STDLIB_HINTS:
        continue
    pip_name = PIP_IMPORT_MAP.get(name)
    if pip_name:
        reqs.append(pip_name)
    else:
        reqs.append(name)

out = []
seen = set()
for r in reqs:
    if r.lower() not in seen:
        out.append(r)
        seen.add(r.lower())
return out

```
def install(self, cancel_event: Optional[threading.Event] = None,
            progress_queue: Optional[queue.Queue] = None) -> Tuple[bool, str]:
    import venv
    if not self.venv_dir.exists():
        if progress_queue:
            progress_queue.put(("status", "Creating virtual environment..."))
        venv.EnvBuilder(with_pip=True, clear=False).create(self.venv_dir)

    py = self.get_python()
    pip = self.venv_dir / ("Scripts/pip.exe" if os.name == "nt" else "bin/pip")

    out_all = []

    if progress_queue:
        progress_queue.put(("status", "Upgrading pip..."))

    rc, out, cancelled = _run([str(py), "-m", "pip", "install", "--upgrade",
                               "pip", "setuptools", "wheel"],
                              cancel_event=cancel_event)
    out_all.append(out)
    if cancelled:
        return False, "\n".join(out_all)

    dep_files = _detect_dep_files(self.tool_root)

    if "requirements.txt" in dep_files:
        if progress_queue:
            progress_queue.put(("status", "Installing from requirements.txt..."))
        rc, out, cancelled = _run([str(pip), "install", "-r",
                                   str(dep_files["requirements.txt"])],
                                  cwd=self.tool_root, cancel_event=cancel_event)
        out_all.append(out)

    elif "pyproject.toml" in dep_files:
        if progress_queue:
            progress_queue.put(("status", "Installing from pyproject.toml..."))
        rc, out, cancelled = _run([str(py), "-m", "pip", "install", "-e", "."],
                                  cwd=self.tool_root, cancel_event=cancel_event)
        out_all.append(out)
        if rc != 0 and not cancelled:
            inferred = _infer_requirements(self.tool_root)
            if inferred:
                if progress_queue:
                    progress_queue.put(("status", "Falling back to inferred deps..."))
                rc, out, cancelled = _run([str(pip), "install"] + inferred,
                                         cwd=self.tool_root, cancel_event=cancel_event)
                out_all.append(out)

    elif "setup.py" in dep_files or "setup.cfg" in dep_files:
        if progress_queue:
            progress_queue.put(("status", "Installing from setup.py..."))
        rc, out, cancelled = _run([str(py), "-m", "pip", "install", "-e", "."],
                                  cwd=self.tool_root, cancel_event=cancel_event)
        out_all.append(out)
        if rc != 0 and not cancelled:
            inferred = _infer_requirements(self.tool_root)
            if inferred:
                rc, out, cancelled = _run([str(pip), "install"] + inferred,
                                         cwd=self.tool_root, cancel_event=cancel_event)
                out_all.append(out)
    else:
        inferred = _infer_requirements(self.tool_root)
        if inferred:
            if progress_queue:
                progress_queue.put(("status", f"Inferred {len(inferred)} dependencies..."))
            req_file = self.tool_root / "requirements.inferred.txt"
            req_file.write_text("\n".join(inferred) + "\n", encoding="utf-8")
            rc, out, cancelled = _run([str(pip), "install", "-r", str(req_file)],
                                      cwd=self.tool_root, cancel_event=cancel_event)
            out_all.append(out)

    return rc == 0, "\n".join(out_all)

def get_python(self) -> Optional[Path]:
    py = self.venv_dir / ("Scripts/python.exe" if os.name == "nt" else "bin/python")
    return py if py.exists() else None

class PoetryInstaller(BaseInstaller):
def check_available(self) -> bool:
rc, _, _ = _run([“poetry”, “–version”], timeout=10)
return rc == 0

```
def install(self, cancel_event: Optional[threading.Event] = None,
            progress_queue: Optional[queue.Queue] = None) -> Tuple[bool, str]:
    out_all = []
    has_lock = (self.tool_root / "poetry.lock").exists()

    if has_lock:
        if progress_queue:
            progress_queue.put(("status", "Installing from poetry.lock..."))
        rc, out, cancelled = _run(["poetry", "install", "--no-interaction"],
                                  cwd=self.tool_root, cancel_event=cancel_event)
        out_all.append("Using poetry.lock for reproducible install...")
    else:
        if (self.tool_root / "pyproject.toml").exists():
            if progress_queue:
                progress_queue.put(("status", "Installing from pyproject.toml..."))
            rc, out, cancelled = _run(["poetry", "install", "--no-interaction"],
                                      cwd=self.tool_root, cancel_event=cancel_event)
            out_all.append("Installing from pyproject.toml...")
        else:
            if progress_queue:
                progress_queue.put(("status", "Initializing poetry project..."))
            rc, out, cancelled = _run(["poetry", "init", "--no-interaction", "--name", "tool"],
                                      cwd=self.tool_root, cancel_event=cancel_event)
            out_all.append(out)

            if not cancelled:
                inferred = _infer_requirements(self.tool_root)
                for i, pkg in enumerate(inferred[:20]):
                    if progress_queue:
                        progress_queue.put(("status", f"Adding {pkg} ({i+1}/{min(20, len(inferred))})..."))
                    rc, out, cancelled = _run(["poetry", "add", pkg],
                                              cwd=self.tool_root, timeout=60,
                                              cancel_event=cancel_event)
                    out_all.append(out)
                    if rc != 0 or cancelled:
                        break

    out_all.append(out)
    return rc == 0, "\n".join(out_all)

def get_python(self) -> Optional[Path]:
    rc, out, _ = _run(["poetry", "env", "info", "--path"], cwd=self.tool_root)
    if rc == 0:
        env_path = Path(out.strip())
        py = env_path / ("Scripts/python.exe" if os.name == "nt" else "bin/python")
        return py if py.exists() else None
    return None
```

def install_with_backend(key: str, backend: Optional[str] = None,
cancel_event: Optional[threading.Event] = None,
progress_queue: Optional[queue.Queue] = None) -> dict:
reg = _load_registry()
if key not in reg[“tools”]:
raise SystemExit(f”Unknown tool key: {key}. Run ‘add’ first.”)

```
t = reg["tools"][key]
tool_root = Path(t["root_dir"])
tool_dir = Path(t["tool_dir"])

if not backend:
    backend = reg["preferences"].get("default_backend", "pip")
    dep_files = _detect_dep_files(tool_root)
    if "poetry.lock" in dep_files:
        backend = "poetry"
    elif "uv.lock" in dep_files:
        backend = "uv"

installer = get_installer(backend, tool_root, tool_dir)
if not installer:
    if progress_queue:
        progress_queue.put(("status", f"{backend} not available, falling back to pip..."))
    installer = PipInstaller(tool_root, tool_dir)
    backend = "pip"

success, output = installer.install(cancel_event=cancel_event,
                                    progress_queue=progress_queue)

result = {
    "backend": backend,
    "success": success,
    "output_tail": output[-4000:],
}

t["last_install"] = result
_save_registry(reg)
return result
```

def run_with_backend(key: str, args: List[str], entry: Optional[str] = None,
backend: Optional[str] = None,
cancel_event: Optional[threading.Event] = None) -> dict:
reg = _load_registry()
if key not in reg[“tools”]:
raise SystemExit(f”Unknown tool key: {key}. Run ‘add’ first.”)

```
t = reg["tools"][key]
tool_root = Path(t["root_dir"])
tool_dir = Path(t["tool_dir"])

if not backend:
    backend = t.get("last_install", {}).get("backend", "pip")

installer = get_installer(backend, tool_root, tool_dir)
if not installer:
    installer = PipInstaller(tool_root, tool_dir)

py = installer.get_python()
if not py or not py.exists():
    return {"returncode": 1, "output_tail": "Python not found. Run install first."}

ep = None
if entry:
    ep_candidate = Path(entry).expanduser()
    ep = (tool_root / ep_candidate).resolve() if not ep_candidate.is_absolute() else ep_candidate.resolve()
else:
    ep = (Path(t["entrypoint"]) if t.get("entrypoint") else None)

if ep is None:
    ep = _guess_entrypoint(tool_root)

if ep is None:
    return {"returncode": 2, "output_tail": "No entrypoint detected."}

if args:
    run_args = args
else:
    looks_cli = _looks_like_cli(ep)
    run_args = ["--help"] if looks_cli else []

rc, out, cancelled = _run([str(py), str(ep), *run_args],
                          cwd=tool_root, cancel_event=cancel_event)

if cancelled:
    return {"returncode": -1, "output_tail": out, "cancelled": True}

if rc != 0:
    healed, heal_msg = _auto_heal_missing_paths(tool_root, out)
    if healed:
        rc2, out2, _ = _run([str(py), str(ep), *run_args], cwd=tool_root)
        out = out + f"\n---auto-heal: {heal_msg}---\n" + out2
        rc = rc2

if rc != 0 and (not args) and run_args == ["--help"]:
    rc2, out2, _ = _run([str(py), str(ep)], cwd=tool_root)
    out = out + "\n---fallback run---\n" + out2
    rc = rc2

result = {"returncode": rc, "output_tail": out[-4000:]}
t["last_smoke_test"] = result
_save_registry(reg)
return result
```

def check_queue(self):
    try:
        while True:
            msg_type, data = self.progress_queue.get_nowait()

            if msg_type == "status":
                self.status_var.set(data)
            elif msg_type == "log":
                self.log(data.get("message", ""), tag=data.get("tag"))
            elif msg_type == "complete":
                self.operation_complete(data)
            elif msg_type == "error":
                self.operation_error(data)

    except queue.Empty:
        pass

    self.after_id = self.root.after(100, self.check_queue)

def start_operation(self, target, operation_name: str):
    if self.worker_thread and self.worker_thread.is_alive():
        messagebox.showwarning("Busy", "Another operation is in progress.")
        return

    self.cancel_event = threading.Event()

    # Safe queue drain
    try:
        while True:
            self.progress_queue.get_nowait()
    except queue.Empty:
        pass

    self.set_ui_busy(True, operation_name)
    self.progress_frame.pack(fill="x", pady=(0, 5))
    self.progress_bar.start(15)

    def wrapper():
        try:
            result = target(self.cancel_event, self.progress_queue)
            if self.cancel_event.is_set():
                self.progress_queue.put(("complete", {"cancelled": True, "result": result}))
            else:
                self.progress_queue.put(("complete", {"cancelled": False, "result": result}))
        except Exception as e:
            self.progress_queue.put(("error", str(e)))

    self.worker_thread = threading.Thread(target=wrapper, daemon=True)
    self.worker_thread.start()

def cancel_operation(self):
    if self.cancel_event and not self.cancel_event.is_set():
        self.status_var.set("Cancelling...")
        self.cancel_event.set()

def operation_complete(self, data: dict):
    self.progress_bar.stop()
    self.progress_frame.pack_forget()
    self.set_ui_busy(False)

    if data.get("cancelled"):
        self.log("\n--- Operation cancelled by user ---\n", tag="warning")
        self.status_var.set("Cancelled")
    else:
        result = data.get("result", {})
        if isinstance(result, dict):
            if result.get("success"):
                self.status_var.set("Completed successfully")
            elif result.get("returncode") is not None:
                self.status_var.set(f"Exit code: {result['returncode']}")
            else:
                self.status_var.set("Completed")
    self.refresh_tools_list()

def operation_error(self, error: str):
    self.progress_bar.stop()
    self.progress_frame.pack_forget()
    self.set_ui_busy(False)
    self.log(f"\n❌ ERROR: {error}\n", tag="error")
    self.status_var.set("Error occurred")
    messagebox.showerror("Error", str(error))

        icon="warning"

        return

    try:
        reg = _load_registry()
        if tool["key"] not in reg["tools"]:
            raise ValueError("Tool not found in registry")

        tool_record = reg["tools"][tool["key"]]
        tool_dir = Path(tool_record["tool_dir"])

        tools_dir_resolved = TOOLS_DIR.resolve()

        del reg["tools"][tool["key"]]
        _save_registry(reg)

        self.log(f"✓ Removed: {tool['name']}", tag="success")

        run_args = run_args[1:]
    res = run_with_backend(args.key, run_args, entry=args.entry, backend=args.backend)
    print(json.dumps(res, indent=2))
    return

if args.cmd == "doctor":
    print(json.dumps(doctor(args.key), indent=2, default=str))
    return

if args.cmd == "pack":
    outp = pack(args.key, args.out)
    print(json.dumps({"packed": True, "out": str(outp)}, indent=2))
    return

if args.cmd == "config":
    if args.default_backend:
        set_default_backend(args.default_backend)
    return

if args.cmd == "gui":
    if not GUI_AVAILABLE:
        print("Error: GUI not available. Install tkinter support.")
        sys.exit(1)
    app = ToolsmithGUI()
    app.run()
    return
